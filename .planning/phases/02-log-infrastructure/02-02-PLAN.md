    ```python
    """Stdlib logging configuration for StoneReader (D-16).

    Call configure_logging() exactly once at app entry (stonereader/__main__.py).
    """
    from __future__ import annotations

    import logging
    import logging.handlers
    import os
    from pathlib import Path

    LOG_DIR = Path.home() / ".stonereader"
    LOG_FILE_NAME = "stonereader.log"


    def configure_logging() -> None:
        """Configure root logger. Idempotent — safe to call more than once."""
        LOG_DIR.mkdir(exist_ok=True)
        level = logging.DEBUG if os.environ.get("STONEREADER_DEBUG") == "1" else logging.INFO

        fmt = logging.Formatter(
            "%(asctime)s %(name)s [%(levelname)s] %(message)s",
            datefmt="%Y-%m-%d %H:%M:%S",
        )

        root = logging.getLogger()
        root.setLevel(level)

        log_path = LOG_DIR / LOG_FILE_NAME
        if not any(isinstance(h, logging.handlers.RotatingFileHandler) for h in root.handlers):
            file_handler = logging.handlers.RotatingFileHandler(
                log_path, maxBytes=2_000_000, backupCount=3, encoding="utf-8"
            )
            file_handler.setFormatter(fmt)
            root.addHandler(file_handler)

        if not any(
            isinstance(h, logging.StreamHandler)
            and not isinstance(h, logging.handlers.RotatingFileHandler)
            for h in root.handlers
        ):
            console_handler = logging.StreamHandler()
            console_handler.setFormatter(fmt)
            root.addHandler(console_handler)
    ```

    ```python
    """Hearthstone log.config bootstrap (D-11).

    Idempotently ensures the [Power] section is present so Hearthstone writes Power.log.
    Preserves all other sections written by other trackers (HDT, Firestone) per Pitfall 5.
    """
    from __future__ import annotations

    import configparser
    import logging
    import os
    from pathlib import Path
    from typing import Optional

    logger = logging.getLogger(__name__)

    REQUIRED_POWER_SECTION = {
        "LogLevel": "1",
        "FilePrinting": "True",
        "ConsolePrinting": "False",
        "ScreenPrinting": "False",
        "Verbose": "True",
    }


    def log_config_path() -> Path:
        """Return the canonical Windows path %LOCALAPPDATA%\\Blizzard\\Hearthstone\\log.config."""
        local_appdata = os.environ.get("LOCALAPPDATA")
        if not local_appdata:
            # Non-Windows or LOCALAPPDATA unset — return best-effort fallback so callers can still pass an override.
            local_appdata = str(Path.home() / "AppData" / "Local")
        return Path(local_appdata) / "Blizzard" / "Hearthstone" / "log.config"


    def ensure_log_config(path: Optional[Path] = None) -> bool:
        """Write or update [Power] section in log.config. Returns True if file changed.

        - Idempotent: second call with already-correct file returns False.
        - Preserves all other sections (HDT, Firestone, etc.).
        - Uses RawConfigParser (no % interpolation — Pitfall 5).
        - Preserves key case via optionxform = str.
        """
        path = path or log_config_path()
        path.parent.mkdir(parents=True, exist_ok=True)

        parser = configparser.RawConfigParser()
        parser.optionxform = str  # type: ignore[assignment]  # preserve key case
        if path.exists():
            parser.read(path, encoding="utf-8")

        changed = False
        if not parser.has_section("Power"):
            parser.add_section("Power")
            changed = True
        for key, value in REQUIRED_POWER_SECTION.items():
            if parser.get("Power", key, fallback=None) != value:
                parser.set("Power", key, value)
                changed = True

        if changed:
            with path.open("w", encoding="utf-8") as f:
                parser.write(f)
            logger.info("Updated log.config at %s", path)
        return changed
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| StoneReader -> %LOCALAPPDATA%\Blizzard\Hearthstone\log.config | Shared INI file with HDT, Firestone, official Hearthstone client. Concurrent writes from us + Hearthstone are theoretically possible (Hearthstone may write log.config during startup). |
| StoneReader -> ~/.stonereader/stonereader.log | Sole writer; no shared state. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-2-02 | Tampering | _log_config.ensure_log_config writes user-shared INI | mitigate | RawConfigParser preserves other sections (HDT, Firestone). optionxform=str preserves Hearthstone's case-sensitive key names. Pitfall 5 (RESEARCH.md line 737) is the blocking acceptance criterion. Idempotent — only writes when delta exists, minimizing concurrent-write race window. |
| T-2-02b | Race condition | log.config concurrent write vs Hearthstone | accept | We write log.config exactly once at app launch (before tracker.start()), and Hearthstone reads it only on its OWN launch. Window for concurrent write is small (user would have to launch StoneReader and Hearthstone in <100ms of each other AND Hearthstone would have to be writing log.config at that exact instant — rare). If race occurs, worst case is log.config is missing one tool's keys; user re-launches and we re-merge. No data loss. v1 acceptable. |
| T-2-04 | Information disclosure | stonereader.log writes user filesystem paths | accept | The log file path is `~/.stonereader/stonereader.log`, owned by the user. No PII (BattleTags, IP) written by Phase 2 modules — only file paths and event types. Phase 3 should review this when adding speech-text logging. |
| T-2-DOS | DoS | Logging file growth | mitigate | RotatingFileHandler caps each file at 2MB with 3 backups = 8MB total. Cannot grow unbounded. |

This plan covers log.config tampering surface (T-2-02). Path-traversal threats for `_log_path.py` are owned by Plan 03; buffer-overflow threats for `_watcher.py` are owned by Plan 06; subscriber-isolation threats for `_tracker.py` are owned by Plan 07.
</threat_model>
